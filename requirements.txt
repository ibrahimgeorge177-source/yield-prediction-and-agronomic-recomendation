# Runtime dependencies for the prediction API (api/).
#
# pandas is pinned below 3.0 deliberately. The pipeline resolves its
# categorical columns with `dtype == object` (scripts/district_model.py,
# load_schema); on pandas 3 text columns arrive as `str`, that test returns
# nothing, and target encoding silently degrades. The model was trained on
# pandas 2.x -- serve it on pandas 2.x.
fastapi>=0.115,<1.0
uvicorn[standard]>=0.30,<1.0
pydantic>=2.7,<3.0

pandas>=2.1,<3.0
numpy>=1.26,<3.0
scipy>=1.11,<2.0
# Pinned, not ranged. A pickled scikit-learn estimator is only guaranteed to
# load in the version that fitted it: the shipped artefact was fitted with
# 1.6.1, and on 1.7.x its SimpleImputer unpickles without the _fill_dtype
# attribute that version expects, so every prediction fails with
# AttributeError at transform time. Keep this in step with the environment
# that runs scripts/train_district_model.py.
scikit-learn==1.6.1
joblib>=1.3,<2.0
lightgbm>=4.1,<5.0
xgboost>=2.0,<4.0

# The shipped ensemble includes an entity-embedding neural member, so the
# artefact only unpickles with torch installed. If your artefact was built with
# `train_district_model.py --no-nn`, this line can be dropped.
torch>=2.2,<3.0
